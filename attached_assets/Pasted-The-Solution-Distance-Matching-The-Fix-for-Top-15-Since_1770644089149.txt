def find_best_pair(self, data):
    # data is a DataFrame with columns like 'BTC/USDT', 'ETH/USDT', etc.
    
    # 1. CLEANUP: Remove stablecoins
    valid_cols = [c for c in data.columns if not any(x in c for x in ['USDT', 'USDC', 'DAI', 'FDUSD'])]
    df = data[valid_cols]
    
    best_pair = None
    highest_corr = -1.0
    
    # 2. SCAN for Highest Correlation
    cols = df.columns
    for i in range(len(cols)):
        for j in range(i + 1, len(cols)):
            a = cols[i]
            b = cols[j]
            
            # Simple Correlation (Last 50 periods)
            corr = df[a].rolling(50).corr(df[b]).iloc[-1]
            
            if corr > highest_corr:
                highest_corr = corr
                best_pair = (a, b)
                
    # 3. VERIFY it's "Good Enough" (e.g., > 0.80)
    if highest_corr > 0.80:
        return best_pair, highest_corr
    else:
        return None, 0.0
