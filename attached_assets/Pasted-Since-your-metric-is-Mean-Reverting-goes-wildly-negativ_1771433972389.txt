  private static history: number[] = [];
  private static readonly WINDOW_SIZE = 60; // 1 hour rolling window

  static addMetric(value: number) {
    this.history.push(value);
    if (this.history.length > this.WINDOW_SIZE) this.history.shift();
  }

  static getStats() {
    if (this.history.length < 10) return { zScore: 0, mean: 0 };

    const sum = this.history.reduce((a, b) => a + b, 0);
    const mean = sum / this.history.length;
    const variance = this.history.reduce((a, b) => a + Math.pow(b - mean, 2), 0) / this.history.length;
    const stdDev = Math.sqrt(variance);

    const current = this.history[this.history.length - 1];
    const zScore = stdDev === 0 ? 0 : (current - mean) / stdDev;

    return { zScore, mean, stdDev, current };
  }
}
