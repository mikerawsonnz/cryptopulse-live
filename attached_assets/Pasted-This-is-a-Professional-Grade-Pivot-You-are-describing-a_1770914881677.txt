export async function executeBasket(probability: number, totalCapital: number) {
  
  // 1. Determine Bias
  // e.g. Prob = 70% -> Long 70%, Short 30%
  const longAllocation = totalCapital * (probability / 100);
  const shortAllocation = totalCapital * ((100 - probability) / 100);

  // 2. Sort the Universe (Relative Strength)
  const rankedCoins = await getRankedTop15(); 
  // Returns [Strongest, ..., Weakest]

  // 3. Select Teams
  // Buy Top 5 Strongest
  const longTeam = rankedCoins.slice(0, 5);
  // Short Bottom 3 Weakest (Concentrate the short bet)
  const shortTeam = rankedCoins.slice(-3);

  // 4. Execute LONGS
  const longSizePerCoin = longAllocation / longTeam.length;
  for (const coin of longTeam) {
    await placeOrder(coin.symbol, 'BUY', longSizePerCoin); // ATR Adjusted size inside placeOrder
  }

  // 5. Execute SHORTS
  const shortSizePerCoin = shortAllocation / shortTeam.length;
  for (const coin of shortTeam) {
    await placeOrder(coin.symbol, 'SELL', shortSizePerCoin);
  }
}

export async function checkBasketExit() {
  const positions = await getAllOpenPositions();
  
  if (positions.length === 0) return;

  // Calculate Net PnL of the Basket
  const totalPnL = positions.reduce((sum, p) => sum + p.unrealizedPnL, 0);
  const totalInvested = positions.reduce((sum, p) => sum + p.marginUsed, 0);
  
  const pnlPercent = (totalPnL / totalInvested) * 100;

  console.log(`🧺 BASKET STATUS: ${pnlPercent.toFixed(2)}% PnL`);

  // WINNER: +1% Target
  if (pnlPercent >= 1.0) {
    console.log(`💰 BASKET TARGET HIT (+1%). Closing All.`);
    await closeAllPositions();
    await enterCooldown(60); // Rest for 1 hour
  }

  // LOSER: -2.5% Stop
  if (pnlPercent <= -2.5) {
    console.log(`🛑 BASKET STOP HIT (-2.5%). Closing All.`);
    await closeAllPositions();
    await enterCooldown(120); // Penalty box for 2 hours
  }
}
