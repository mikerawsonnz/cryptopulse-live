import time
import os
import json

LOCK_FILE = "trade_lock.json"

def can_trade(symbol):
    # 1. Load Lock File
    if not os.path.exists(LOCK_FILE):
        with open(LOCK_FILE, 'w') as f: json.dump({}, f)
    
    try:
        with open(LOCK_FILE, 'r') as f: locks = json.load(f)
    except:
        locks = {}

    last_time = locks.get(symbol, 0)
    
    # 2. Check Time (5 Minutes = 300s)
    if time.time() - last_time < 300: 
        return False
    
    return True

def register_trade(symbol):
    # 3. Write Lock Immediately
    try:
        with open(LOCK_FILE, 'r') as f: locks = json.load(f)
    except:
        locks = {}
        
    locks[symbol] = time.time()
    
    with open(LOCK_FILE, 'w') as f: json.dump(locks, f)
