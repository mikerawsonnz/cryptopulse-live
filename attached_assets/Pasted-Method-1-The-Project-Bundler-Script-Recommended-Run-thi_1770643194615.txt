Method 1: The "Project Bundler" Script (Recommended)
Run this simple Python script in your Replit/Project root folder. It will scan your project, ignore junk files (like virtual environments or data logs), and combine all your code into one file called full_project_code.txt.

1. Create a file named bundler.py and paste this code:

Python
import os

# Files and folders to IGNORE (Safety & Noise filter)
IGNORE_DIRS = {'.git', '__pycache__', 'venv', 'env', '.replit', '.upm', 'data', 'logs'}
IGNORE_FILES = {'poetry.lock', 'pyproject.toml', 'bundler.py', 'README.md', '.env'}
# Only include these extensions
INCLUDE_EXT = {'.py', '.json', '.txt', '.md'}

def bundle_project():
    output_file = "full_project_code.txt"
    
    with open(output_file, "w", encoding="utf-8") as outfile:
        # Write the Directory Tree first (for context)
        outfile.write("--- PROJECT STRUCTURE ---\n")
        for root, dirs, files in os.walk("."):
            # Filter directories
            dirs[:] = [d for d in dirs if d not in IGNORE_DIRS]
            level = root.replace(".", "").count(os.sep)
            indent = " " * 4 * (level)
            outfile.write(f"{indent}{os.path.basename(root)}/\n")
            subindent = " " * 4 * (level + 1)
            for f in files:
                if f not in IGNORE_FILES and os.path.splitext(f)[1] in INCLUDE_EXT:
                    outfile.write(f"{subindent}{f}\n")
        
        outfile.write("\n\n--- FILE CONTENTS ---\n")

        # Walk again to write content
        for root, dirs, files in os.walk("."):
            dirs[:] = [d for d in dirs if d not in IGNORE_DIRS]
            
            for file in files:
                if file in IGNORE_FILES: continue
                if os.path.splitext(file)[1] not in INCLUDE_EXT: continue
                
                file_path = os.path.join(root, file)
                
                try:
                    outfile.write(f"\n{'='*20}\nSTART OF FILE: {file_path}\n{'='*20}\n")
                    with open(file_path, "r", encoding="utf-8") as infile:
                        content = infile.read()
                        # REDACT API KEYS (Basic security check)
                        if "api_key" in content.lower() or "secret" in content.lower():
                            outfile.write("# [NOTE: POTENTIAL SENSITIVE DATA REDACTED BY USER BEFORE UPLOAD]\n")
                        outfile.write(content)
                    outfile.write(f"\nEND OF FILE: {file_path}\n")
                except Exception as e:
                    outfile.write(f"\nCould not read {file_path}: {e}\n")

    print(f"✅ Success! Upload '{output_file}' to the chat.")

if __name__ == "__main__":
    bundle_project()
2. Run the script: python bundler.py
