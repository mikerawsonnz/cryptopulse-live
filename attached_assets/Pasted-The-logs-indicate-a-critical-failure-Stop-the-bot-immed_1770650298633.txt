import os
import json
import time

LOCK_FILE = "trade_lock.json"

def register_trade(symbol):
    try:
        # Read existing
        if os.path.exists(LOCK_FILE):
            with open(LOCK_FILE, 'r') as f: locks = json.load(f)
        else:
            locks = {}
            
        locks[symbol] = time.time()
        
        # WRITE AND FLUSH
        with open(LOCK_FILE, 'w') as f: 
            json.dump(locks, f)
            f.flush()      # Force write to buffer
            os.fsync(f.fileno()) # Force write to disk
            
        print(f"🔒 LOCKED: {symbol} for 5 minutes")
    except Exception as e:
        print(f"Lock Error: {e}")
