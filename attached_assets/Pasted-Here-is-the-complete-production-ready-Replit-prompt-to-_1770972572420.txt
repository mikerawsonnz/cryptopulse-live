export class TimeframeManager {
  /**
   * Calculates the optimal timeframe based on market volatility.
   * "The Gearbox"
   */
  static async getDynamicTimeframe(): Promise<string> {
    // 1. Get Top 15 Volatility Data
    const coins = await db.query.marketData.findMany({
      limit: 15,
      orderBy: [desc(marketData.volume24h)]
    });

    if (!coins.length) return '1h'; // Default safe

    // 2. Calculate Market Volatility Ratio
    let totalRatio = 0;
    let count = 0;

    for (const coin of coins) {
      if (coin.atr24h > 0) {
        totalRatio += (coin.currentAtr / coin.atr24h);
        count++;
      }
    }
    
    const marketVolRatio = count > 0 ? totalRatio / count : 1.0;
    console.log(`⚙️ GEARBOX: Volatility Ratio is ${marketVolRatio.toFixed(2)}x`);

    // 3. Select Gear
    if (marketVolRatio > 2.0) {
      console.log("⚡ GEAR: 5-Minute (Extreme Vol)");
      return '5m';
    } 
    else if (marketVolRatio > 1.2) {
      console.log("🏎️ GEAR: 15-Minute (Trending)");
      return '15m';
    } 
    else if (marketVolRatio < 0.8) {
      console.log("🐢 GEAR: 4-Hour (Chop/Dead)");
      return '4h';
    } 
    
    console.log("🚗 GEAR: 1-Hour (Standard)");
    return '1h';
  }
}

interface BreadthResult {
  score: number;        // 0 to 100
  longRatio: number;    // 0.0 to 1.0
  shortRatio: number;   // 0.0 to 1.0
  regime: 'BULL' | 'BEAR' | 'NEUTRAL';
}

export class MarketBreadth {
  static async analyze(timeframe: string): Promise<BreadthResult> {
    const coins = await db.query.marketData.findMany({ limit: 15 });
    
    let score = 0;

    // 1. STRUCTURE (40 pts): Price vs VWAP
    const aboveVwap = coins.filter(c => c.price > c.vwap).length;
    score += (aboveVwap / 15) * 40;

    // 2. MOMENTUM (30 pts): Dynamic Timeframe Trend
    // Assuming 'changeDynamic' is populated based on the 'timeframe' passed
    const positiveMom = coins.filter(c => c.changeDynamic > 0).length;
    score += (positiveMom / 15) * 30;

    // 3. VOLATILITY EXPANSION (30 pts): ATR Expanding?
    const expanding = coins.filter(c => c.currentAtr > c.atr24h).length;
    if (expanding > 8) score += 30; // Majority expanding

    // 4. Calculate Ratios
    const longRatio = Math.max(0.1, Math.min(0.9, score / 100)); // Clamp between 10% and 90%
    
    let regime: BreadthResult['regime'] = 'NEUTRAL';
    if (score > 65) regime = 'BULL';
    if (score < 35) regime = 'BEAR';

    console.log(`🌊 BREADTH: Score ${score.toFixed(0)} | Regime: ${regime}`);
    
    return {
      score,
      longRatio,
      shortRatio: 1 - longRatio,
      regime
    };
  }
}

import { MarketBreadth } from "./MarketBreadth";
import { db } from "@db";
import { marketData } from "@db/schema"; // schema

export class BasketExecutor {
  
  static async executeBasketCheck(totalCapital: number) {
    // 1. Get The Gear
    const timeframe = await TimeframeManager.getDynamicTimeframe();

    // 2. Get The Sentiment
    const sentiment = await MarketBreadth.analyze(timeframe);

    // ONLY EXECUTE IF REGIME IS CLEAR (Avoid Chop)
    if (sentiment.regime === 'NEUTRAL') {
      console.log("⏸️ BASKET: Market is Neutral. Holding.");
      return;
    }

    // 3. Rank Coins (Relative Strength)
    const allCoins = await db.query.marketData.findMany({ limit: 15 });
    
    // Sort by RS (Price - EMA) / ATR
    const sorted = allCoins.sort((a, b) => {
      const rsA = (a.price - a.ema20) / a.currentAtr;
      const rsB = (b.price - b.ema20) / b.currentAtr;
      return rsB - rsA; // Descending
    });

    const longTeam = sorted.slice(0, 5); // Strongest
    const shortTeam = sorted.slice(-3);  // Weakest

    console.log(`🧺 BASKET PLAN: Long ${longTeam.map(c=>c.symbol)} | Short ${shortTeam.map(c=>c.symbol)}`);

    // 4. Allocate Capital (Risk Parity Logic)
    const longCap = totalCapital * sentiment.longRatio;
    const shortCap = totalCapital * sentiment.shortRatio;

    const orders = [];

    // Longs
    for (const coin of longTeam) {
      // Risk Parity: Higher ATR = Smaller Size
      // Base calculation: (Allocation / 5) * (AvgATR / CoinATR)
      const size = (longCap / 5); // Simplified for safety, ideally use ATR scaler
      orders.push(executeTrade({ symbol: coin.symbol, side: 'BUY', amount: size, mode: 'BASKET' }));
    }

    // Shorts
    for (const coin of shortTeam) {
      const size = (shortCap / 3);
      orders.push(executeTrade({ symbol: coin.symbol, side: 'SELL', amount: size, mode: 'BASKET' }));
    }

    // 5. Fire All
    await Promise.all(orders);
    console.log("✅ BASKET EXECUTED.");
  }
}
